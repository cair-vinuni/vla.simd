# Dependencies of serve/policy_server.py.
#
#   uv venv .venv-serve
#   uv pip install --python .venv-serve -r serve/requirements.txt
#
# lerobot is here for the wire types only: the async-inference payloads are
# pickled lerobot dataclasses, and the reply carries a real torch.Tensor. No
# policy is built and no device is touched - inference is entirely
# libvla_simd_<model>.so.
#
# SECURITY: pickle over an unauthenticated gRPC port, in the control path of a
# physical arm. Bind to 127.0.0.1 or a trusted LAN and tunnel over SSH/VPN.
#
# Two install gotchas:
#  - protobuf is missing from lerobot's own dependency set; without it the
#    server exits with `No module named 'google'`.
#  - on aarch64 lerobot resolves torch to the CUDA build, pulling ~3.5 GB of
#    nvidia-* wheels. Reinstall CPU torch afterwards, not before - lerobot
#    overwrites it:
#      uv pip install --index-url https://download.pytorch.org/whl/cpu torch

numpy>=2.1,<3
lerobot>=0.6.1,<0.7     # 0.6.0 fixes CVE-2026-25874 in lerobot's own server
grpcio>=1.68,<2
protobuf>=5.29,<8
torch>=2.8,<3
